## =============================================================================
## run_chunk_bci.txt — BCI stem-identification launch reference
## =============================================================================
## All commands are run from the **project root** (the directory that contains
## dp_global/ and BCI_stem_reconstruction/).
##
## Each run creates a timestamped output directory inside BASE_OUT_DIR:
##   BASE_OUT_DIR/<timestamp_run_code>/
## where <timestamp_run_code> encodes the start time and key config flags
## (e.g. 20260516_043718_unknown_allT_DP_MB_NME_g5_sm0p5_kg0_ks0_rcpp).
##
## Per-chunk Feather files land in that directory.  After all chunks finish,
## run 2_merge_chunks_to_datatable.R to merge them into the final dataset.
## =============================================================================


## -----------------------------------------------------------------------------
## START — launch a new full run
## -----------------------------------------------------------------------------
## Run from the project root (the directory that contains dp_global/ and
## BCI_stem_reconstruction/).

Rscript BCI_stem_reconstruction/2_STEM_IDENTIFICATION/1_main_cpp_chunk_bci.R \
    --DP_MAX_STATES=10000 \
    ## Cap on the number of DP state-space entries evaluated per tag.
    ## Higher = more accuracy, longer runtime.  10000 ≈ 30 min/tag maximum
    ## on a 2020+ Mac.  Raise to 20000–50000 for stress-tests.
    \
    --PROB_SPECIES="oenoma,bactma,ficuob,ficupo,ficuc2,ficubu,ficuc1,ficuci,ficupe" \
    ## Comma-separated Mnemonic codes for species that bypass the DP engine
    ## and go directly to the probabilistic greedy matcher.  Used for figs and
    ## other strangler species whose stem topology defeats the DP state space.
    \
    --DP_FALLBACK_GROWTH_FORMS="strangler" \
    ## Growth forms for which the DP engine falls back to the probabilistic
    ## matcher when the state space exceeds DP_MAX_STATES.  "strangler" covers
    ## strangler figs not already routed by PROB_SPECIES.
    \
    --POSTERIOR_SAMPLE_SEED=42 \
    ## Integer seed for reproducible posterior path sampling and probabilistic
    ## Gumbel-noise draws.  Fix to any integer for regression-comparable runs.
    \
    --MANUAL_CORES=TRUE \
    ## When TRUE, use MANUAL_CORES_VALUE instead of auto-detecting CPU count.
    \
    --MANUAL_CORES_VALUE=18 \
    ## Number of parallel workers (mclapply forks).  Set to the number of
    ## physical cores available (18 on the production Mac).
    \
    --DP_CHUNK_SIZE=18 \
    ## Number of tags processed in each parallel chunk.  Matching this to
    ## MANUAL_CORES_VALUE ensures all workers stay busy for most chunks.
    \
    --USE_MEASUREMENT_ERROR=FALSE \
    ## When TRUE, the DP solver uses per-census DBH measurement-error SDs to
    ## widen transition costs.  Set FALSE for a faster, deterministic run.
    \
    --BASE_OUT_DIR=/Users/medinaja/outputs_bci_stem_identification
    ## Root directory for all output.  A subdirectory named by timestamp +
    ## config flags is created automatically on each new run.
    ## Per-chunk Feather files are written inside that subdirectory.
